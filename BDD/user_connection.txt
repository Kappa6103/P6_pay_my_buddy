CREATE TABLE IF NOT EXISTS `mydb`.`user_connection` (
  `userId1` INT NOT NULL,
  `userId2` INT NOT NULL,
  PRIMARY KEY (`userId1`, `userId2`),
  INDEX `fk_user_has_user_user2_idx` (`userId2` ASC) VISIBLE,
  INDEX `fk_user_has_user_user1_idx` (`userId1` ASC) VISIBLE,
  CONSTRAINT `fk_user_has_user_user1`
    FOREIGN KEY (`userId1`)
    REFERENCES `mydb`.`user` (`userId`)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION,
  CONSTRAINT `fk_user_has_user_user2`
    FOREIGN KEY (`userId2`)
    REFERENCES `mydb`.`user` (`userId`)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION)
ENGINE = InnoDB
