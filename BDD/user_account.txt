CREATE TABLE IF NOT EXISTS `mydb`.`user_account` (
  `user_userId` INT NOT NULL,
  `balance` DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (`user_userId`),
  INDEX `fk_user_account_user1_idx` (`user_userId` ASC) VISIBLE,
  CONSTRAINT `fk_user_account_user1`
    FOREIGN KEY (`user_userId`)
    REFERENCES `mydb`.`user` (`userId`)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION)
ENGINE = InnoDB
